func (repl *Replicator) Reset() {
	if !shared.IsReplDisabled {
		repl.txnCache, repl.lastLogClk = make([][]PairClockUpdates, nGoRoutines), make([]clocksi.Timestamp, nGoRoutines)
		dummyTs := clocksi.NewSliceTimestamp()
		for id := 0; id < int(nGoRoutines); id++ {
			cacheEntry := make([]PairClockUpdates, 1, cacheInitialSize)
			//All txns have a clock higher than this, so the first request is the equivalent of "requesting all commited txns"
			cacheEntry[0] = PairClockUpdates{clk: dummyTs}
			repl.txnCache[id] = cacheEntry
		}
		fmt.Println("[REPLICATOR]Reset complete.")
	}
}

func (repl *Replicator) Initialize(tm *TransactionManager, loggers []Logger, buckets []string, replicaID uint16) {
	if !shared.IsReplDisabled {
		if !repl.started {
			repl.tm = tm
			repl.started = true
			repl.localPartitions = loggers
			//bucketsToListen := make([]string, 1)
			//bucketsToListen[0] = "*"
			bucketsToListen := buckets

			//remoteConn, err := CreateRemoteGroupStruct(bucketsToListen, replicaID)
			//TODO: Not ignore err
			//ignore(err)
			//repl.remoteConn = remoteConn
			remoteConn := CreateRemoteGroupStruct(bucketsToListen, replicaID)
			repl.remoteConn = remoteConn
			repl.buckets = bucketsToListen
			//repl.initialDummyTs = clocksi.NewClockSiTimestampFromId(replicaID)
			repl.initialDummyTs = clocksi.NewSliceTimestamp()
			repl.replicaID = replicaID
			repl.partsChan = make(chan StableClkUpdatesPair, nGoRoutines)
			//repl.receiveHold = make(map[uint16]TMRemoteTxn)

			//Also skips the joining algorithm if there's no replica to join
			if doesJoin && len(remoteConn.conns) > 0 {
				fmt.Println("[REPLICATOR]Join mode")
				repl.JoinInfo = JoinInfo{holdMsgs: make([]ReplicatorMsg, 0, joinHoldInitialSize), allDone: false, nHoldJoins: 0}
				repl.joinGroup()
			} else {
				//Wait for replicaIDs of existing replicas
				repl.JoinInfo.waitFor = int(remoteConn.nReplicas)
				crdt.NReplicas = int32(remoteConn.nReplicas + 1)
				fmt.Printf("[REPLICATOR]Not doing join, will wait for replicaID of existing replicas. IDs to receive: %d, number of replicas (including self): %d\n",
					repl.JoinInfo.waitFor, crdt.NReplicas)
				remoteConn.sendReplicaID(bucketsToListen, localPotionIP)
				go repl.receiveRemoteTxns()
				//go repl.replicateCycle()
				//If there's no other replica, we can start right away
				if len(remoteConn.conns) == 0 {
					fmt.Println("[REPLICATOR] PotionDB in single server mode.")
					repl.initialDummyTs = repl.initialDummyTs.Update()
					repl.allDone = true
					go repl.replicateCycle()
					go repl.tm.SendRemoteMsg(TMStart{}) //Different thread to avoid blocking
				}
			}
		}
	} else {
		fmt.Println("[REPLICATOR] Warning - replicator is disabled. PotionDB started in single server mode.")
		go tm.SendRemoteMsg(TMStart{})
	}
}

func (repl *Replicator) replicateCycle() {
	repl.txnCache, repl.lastLogClk = make([][]PairClockUpdates, nGoRoutines), make([]clocksi.Timestamp, nGoRoutines)
	/*for id := 0; id < int(nGoRoutines); id++ {
		cacheEntry := make([]PairClockUpdates, 1, cacheInitialSize)
		//All txns have a clock higher than this, so the first request is the equivalent of "requesting all commited txns"
		cacheEntry[0] = PairClockUpdates{clk: repl.initialDummyTs}
		repl.txnCache[id] = cacheEntry
	}*/
	for id := 0; id < int(nGoRoutines); id++ {
		repl.txnCache[id] = make([]PairClockUpdates, 0, cacheInitialSize)
		repl.lastLogClk[id] = repl.initialDummyTs
	}

	replSinceLastGC := false //Keeps track if there was any new txns since the last call to cleanState()
	count, start, finish, toSleep := time.Duration(0), int64(0), int64(0), time.Duration(0)
	var startFull time.Time
	previousLen := 0
	//fmt.Printf("[REPLICATOR]Starting replicator cycle at %s\n", time.Now().Format("15:04:05.000"))
	for {
		startFull = time.Now()
		start = startFull.UnixNano()
		utilities.FancyDebugPrint(utilities.REPL_PRINT, repl.replicaID, "starting replicateCycle")

		//fmt.Printf("[REPLICATOR]Waiting for getNewTxns() at %s\n", time.Now().Format("15:04:05.000")) //TODO: CHeck if this is happening
		repl.getNewTxns()
		//fmt.Printf("[REPLICATOR]Got new txns to replicate at %s. Number of partitions: %d\n", time.Now().Format("15:04:05.000"), len(repl.localPartitions))

		count++
		toSend := repl.prepareDataToSend()
		if len(toSend) > 0 {
			replSinceLastGC = true
			//fmt.Printf("[REPLICATOR]Sending ops at %s. Started processing at %s. Number of remote txns: %d\n",
			//time.Now().Format("2006-01-02 15:04:05.000"), startFull.Format("2006-01-02 15:04:05.000"), len(toSend))
			fmt.Println("[REPLICATOR]Sending ops.", "Number of remote txns:", len(toSend), "at", time.Now().Format("15:04:05.000"))
			repl.sendTxns(toSend)
			//end := time.Now()
			//fmt.Printf("[REPLICATOR]Finished sending ops at %s. Started at: %s. Total time taken: %d (ms)\n",
			//end.Format("2006-01-02 15:04:05.000"), startFull.Format("2006-01-02 15:04:05.000"), (end.UnixNano()-start)/1000000)
		}
		utilities.FancyDebugPrint(utilities.REPL_PRINT, repl.replicaID, "finishing replicateCycle")

		if replSinceLastGC {
			repl.cleanState()
			replSinceLastGC = false
		}
		finish = time.Now().UnixNano()
		toSleep = tsSendDelay - time.Duration((finish-start)/1000000)
		if toSleep > 10 {
			time.Sleep(toSleep * time.Millisecond)
		}
		if len(toSend) == 0 && (previousLen > 0 || count*tsSendDelay%60000 == 0) {
			fmt.Println("[REPLICATOR]No ops to send.")
		}
		//if len(toSend) == 0 {
		//	fmt.Println("[REPLICATOR]No ops to send.")
		//} else {
		//	fmt.Println("[REPLICATOR]Sent ops. Number of remote txns:", len(toSend))
		//}
		previousLen = len(toSend)
	}
}

func (repl *Replicator) cleanState() {
	for id, partCache := range repl.txnCache {
		repl.txnCache[id] = make([]PairClockUpdates, len(partCache), tools.Max(len(partCache), cacheInitialSize))
		copy(repl.txnCache[id], partCache)
	}
}

func (repl *Replicator) getNewTxns() {
	//Request for new txns that are not yet present in the txnCache. LastLogClk indicates the latest txn known to the Replicator from each partition.
	for id, part := range repl.localPartitions {
		part.SendLoggerRequest(LoggerRequest{LogRequestArgs: LogTxnArgs{lastClock: repl.lastLogClk[id], ReplyChan: repl.partsChan}})
	}

	repl.maxCommonClk = clocksi.HighestTs
	//Receive replies and cache and also determines the latest common (to all partitions) clk
	//fmt.Printf("[REPLICATOR][GetNewTxns]\n")
	for id := uint64(0); id < nGoRoutines; id++ {
		reply := <-repl.partsChan
		cacheEntry := repl.txnCache[int(reply.partID)]
		//fmt.Printf("Part %d: %d updates.\t", reply.partID, len(reply.upds))
		cacheEntry = append(cacheEntry, reply.upds...)
		reply.upds = nil //For GC purposes
		repl.lastLogClk[int(reply.partID)] = reply.stableClock
		repl.txnCache[int(reply.partID)] = cacheEntry
		if reply.stableClock.IsLower(repl.maxCommonClk) {
			repl.maxCommonClk = reply.stableClock
		}
	}
	//fmt.Println()
}

func (repl *Replicator) prepareDataToSend() (toSend []RemoteTxn) {
	//Helper variables for the cycle
	var currClk clocksi.Timestamp
	var partCache []PairClockUpdates
	var firstEntry PairClockUpdates
	var clkCompare clocksi.TsResult
	var partID int

	//We know the maximum common clock so... we need to keep all txns that are <= it.
	//The problem is... we need to group the txns by clock, right?
	//How to navigate the cache in an efficient way? I suppose that was what the stableClk was for :(
	//Possible solutions:
	//1st - iterate the whole array, to determine the min clock. Then, iterate again to gather the txns.
	//2nd - iterate and try to gather; when a lower clock is found, delete (or postpone?) the current cache.
	//Option 1 seems better to avoid creating and deleting stuff. Let's try to cache the partitions that are involved.

	partsForThisClk := tools.NewSliceWithCounter[int](int(nGoRoutines)) //Keeps track of the partitions involved in the current txn.
	toSend = make([]RemoteTxn, 0, toSendInitialSize)
	done := false
	nSkip := 0 //Debug info
	//fmt.Printf("[REPLICATOR][PrepareDataToSend]Starting to prepare data to send. MaxCommonClk: %s\n", repl.maxCommonClk.ToSortedString())
	for !done { //TODO: Think if we can find a txn > maxStableClk. If so, we can't replicate that.
		currClk = clocksi.HighestTs
		for id := 0; id < len(repl.localPartitions); id++ { //1st phase: find the minimum common clock, and collect partitions involved
			if len(repl.txnCache[id]) == 0 { //This partition has no txns left to replicate (e.g. can happen that a partition receives no updates)
				//fmt.Printf("[REPLICATOR][PrepareDataToSend]Partition %d has no txns left to replicate.\n", id)
				nSkip++
				continue
			}
			partCache = repl.txnCache[id]
			firstEntry = partCache[0]
			clkCompare = firstEntry.clk.Compare(currClk)
			if clkCompare == clocksi.LowerTs { //Reset
				//fmt.Printf("[REPLICATOR][PrepareDataToSend]Found a lower clock. Previous: %s, Current: %s\n", currClk.ToSortedString(), firstEntry.clk.ToSortedString())
				partsForThisClk.Clear()
				partsForThisClk.AddToEnd(id)
				currClk = firstEntry.clk
			} else if clkCompare == clocksi.EqualTs { //Add this partition
				//fmt.Printf("[REPLICATOR][PrepareDataToSend]Found an equal clock. Current: %s\n", currClk.ToSortedString())
				partsForThisClk.AddToEnd(id)
			} else { //else: ignore, this partition does not belong to this txn
				//fmt.Printf("[REPLICATOR][PrepareDataToSend]Ignored clock, as it is not lower or equal. Result: %v. NewClk: %s. CurrClk: %s\n",

		}

			//fmt.Printf("[REPLICATOR][PrepareDataToSend]Nothing further to replicate, stopping iteration. Len of toSend: %d\n", len(toSend))
			return
		}

		maxClkComp := currClk.Compare(repl.maxCommonClk)
		if maxClkComp != clocksi.EqualTs && maxClkComp != clocksi.LowerTs { //Higher or concurrent to maxClk, can't include this one
			return
		}

		}

		if maxClkComp == clocksi.EqualTs { //Can stop iterating
			//fmt.Printf("[REPLICATOR][PrepareDataToSend]CurrClk is equal to maxCommonClk, stopping iteration. CurrClk: %s, MaxCommonClk: %s\n",
			//	currClk.ToSortedString(), repl.maxCommonClk.ToSortedString())

func (repl *Replicator) returnBuffersToLog() {
	copySlice := repl.logBuffersToReturn.Copy()
	repl.logBuffersToReturn.Clear()
	for i := 0; i < len(repl.overflowTxnCache); i++ {
		repl.overflowTxnCache[i].ShiftElementsLeft() //This is very cheap (quick), as it is a slice of slices - and the (inside) slices will be very few.
	}
	go repl.returnBuffersToLogHelper(copySlice.ToSlice())
}

func (repl *Replicator) returnBuffersToLogHelper(buffers []tools.Pair[int, tools.SliceWithHideable[PairClockUpdates]]) {
	for _, buffer := range buffers {

		repl.localPartitions[buffer.First].SendLoggerRequest(LoggerRequest{LogRequestArgs: LogBufferReturnArgs{Buf: buffer.Second.ToFullSlice()}})
	}
}

func (repl *Replicator) sendGroupTxns(toSend []RemoteTxn) {
	//fmt.Printf("[REPLICATOR]Sending txns in group (size: %d). Min: %d, max: %d\n", len(toSend), minTxnsToGroup, maxTxnsPerGroup)
	end := 0
	for len(toSend) > 0 {
		end = utilities.MinInt(len(toSend), maxTxnsPerGroup)
		//start, end = currOffset, utilities.MinInt(len(toSend)-currOffset, maxTxnsPerGroup)+currOffset //for txns variable
		currTxns := toSend[:end]
		firstClk := currTxns[0].Clk
		if !currTxns[len(currTxns)-1].Clk.IsEqualExceptForSelf(firstClk, repl.replicaID) { //The last txn depends on txn(s) of other replicas. We search for the cutoff point.
			for i, txn := range currTxns {
				if !txn.Clk.IsEqualExceptForSelf(firstClk, repl.replicaID) {
					//fmt.Printf("[REPLICATOR]Clk of other replica changed. This happened at index %d. Clks. %s, %s\n", i, firstClk.ToSortedString(), txn.Clk.ToSortedString())
					end = i
					currTxns = currTxns[:i]
					//This txn depends on some txn of another replica, thus better not put in this group
					break
				}
			}
		}

		if end < minTxnsToGroup/2 {
			//Send individually, too few transactions.
			//fmt.Printf("[REPLICATOR]Sending individual in groupTxns. End: %d, min/2: %d, nTxns: %d\n", end, minTxnsToGroup, len(currTxns))
			repl.remoteConn.SendTxnsIndividually(currTxns)
		} else {
			//Send the slice
			//fmt.Printf("[REPLICATOR]Sending group in groupTxns. Size: %d\n", len(currTxns))
			repl.remoteConn.SendGroupTxn(currTxns)
		}
		toSend = toSend[end:] //Hide positions that were already sent
	}
}

func (repl *Replicator) handleJoin(req *Join) {
	joinChan := make(chan TimestampPartIdPair, len(repl.localPartitions))
	repl.tm.mat.SendRequestToAllChannels(MaterializerRequest{MatCommitedClkArgs{ReplyChan: joinChan}})
	replyJoin := ReplyJoin{SenderID: repl.replicaID, CommonBkts: req.CommonBkts,
		Clks: make([]clocksi.Timestamp, len(repl.localPartitions)), ReqIP: localRabbitMQIP}

	sendTo := repl.remoteConn.AddReplica(req.ReqIP, repl.buckets, req.SenderID)
	for range repl.localPartitions {
		reply := <-joinChan
		replyJoin.Clks[reply.partID] = reply.Timestamp
	}
	repl.tm.SendRemoteMsg(TMReplicaID{ReplicaID: req.SenderID})
	repl.remoteConn.SendReplyJoin(replyJoin, sendTo)
}

func (repl *Replicator) handleRequestBucket(req *RequestBucket) {
	bktMap := make(map[string]struct{}, len(req.Buckets))
	for _, bkt := range req.Buckets {
		bktMap[bkt] = struct{}{}
	}
	tmRequest := TMGetSnapshot{Buckets: bktMap, ReplyChan: make(chan TMGetSnapshotReply)}
	repl.tm.SendRemoteMsg(tmRequest)
	snapshotReply := <-tmRequest.ReplyChan
	replyBucket := ReplyBucket{SenderID: repl.replicaID, PartStates: snapshotReply.PartStates, Clk: snapshotReply.Timestamp}
	repl.remoteConn.SendReplyBucket(replyBucket, req.ReqIP)
}

func (repl *Replicator) handleReplyJoin(req *ReplyJoin) {
	fmt.Println("Handling replyJoin")
	repl.holdReplyJoins[repl.nHoldJoins] = req
	repl.waitFor--
	repl.nHoldJoins++
	repl.tm.SendRemoteMsg(TMReplicaID{ReplicaID: req.SenderID})
	if repl.waitFor == 0 {
		//Got all the replies, so we can now ask for the buckets
		repl.askBuckets()
	}
}

func (repl *Replicator) handleReplyBucket(req *ReplyBucket) {
	//Need to count how many were received in order to, at the end, return to the "normal replicator state"
	repl.tm.SendRemoteMsg(TMApplySnapshot{Timestamp: req.Clk, PartStates: req.PartStates})
	repl.waitFor--
	if repl.waitFor == 0 {
		repl.allDone = true
	}
}

func (repl *Replicator) handleReplyEmpty() {
	repl.waitFor--
	if repl.waitFor == 0 && repl.nHoldJoins == 0 {
		//Everyone replied empty. Thus, no need to ask for buckets
		repl.allDone = true
		fmt.Println("ReplyEmpty, setting all done to true")
	} else if repl.waitFor == 0 {
		fmt.Println("Hold joins:", len(repl.holdReplyJoins), repl.holdReplyJoins)
		//At least one of the replies wasn't empty, thus we should ask for buckets
		repl.askBuckets()
	}
	fmt.Println("ReplyEmpty")
}

	repl.remoteConn.SendReplyEmpty(sendTo)
}

func (repl *Replicator) joinGroup() {
	fmt.Println("joinGroup")
	go repl.queueRemoteRequests()
	repl.waitFor = len(repl.remoteConn.conns)
	crdt.NReplicas = int32(repl.waitFor + 1)
	repl.holdReplyJoins = make([]*ReplyJoin, repl.waitFor)
	fmt.Println("Requesting remoteGroup to send join")
	repl.remoteConn.SendJoin(repl.buckets, repl.replicaID)
}

// While the replica is in joining process, this method is used to queue any non-join related msg to a queue.
// Join msgs are processed differently depending on its type.
func (repl *Replicator) queueRemoteRequests() {
	var msg ReplicatorMsg
	for !repl.allDone {
		msg = repl.remoteConn.GetNextRemoteRequest()
		switch typedMsg := msg.(type) {
		case *ReplyJoin:
			fmt.Println("[REPL]ReplyJoin")
			repl.handleReplyJoin(typedMsg)
		case *ReplyBucket:
			fmt.Println("[REPL]ReplyBucket")
			repl.handleReplyBucket(typedMsg)
		case *Join:
			fmt.Println("[REPL]Join")
			//fmt.Println(typedMsg.ReplyID)
			//repl.remoteConn.SendReplyEmpty(typedMsg.ReplyID)
			repl.createConnAndReplyEmpty(typedMsg)
		case *ReplyEmpty:
			fmt.Println("[REPL]ReplyEmpty")
			//Another replica that is also joining
			repl.handleReplyEmpty()
		default:
			fmt.Println("[REPL]Default")
			repl.holdMsgs = append(repl.holdMsgs, msg)
		}
	}
	fmt.Println("Leaving queueRemoteRequests")
	repl.tm.SendRemoteMsg(TMStart{})
	//Handle messages on hold
	for _, holdReq := range repl.holdMsgs {
		repl.handleRemoteRequest(holdReq)
	}
	repl.JoinInfo = JoinInfo{}
	go repl.receiveRemoteTxns()
	go repl.replicateCycle()
}

func (repl *Replicator) askBuckets() {
	//For each bucket, choose the replica with highest clock.
	//If there's no clock for a bucket, then we can assume that bucket has no content yet anywhere.
	bestBktClks := make(map[string]clocksi.Timestamp)
	askTo := make(map[string]string) //bucket -> ip
	repl.holdReplyJoins = repl.holdReplyJoins[0:repl.nHoldJoins]
	for _, replyJoin := range repl.holdReplyJoins {
		for i, clk := range replyJoin.Clks {
			bkt := replyJoin.CommonBkts[i]
			if clk.IsHigher(bestBktClks[bkt]) {
				bestBktClks[bkt] = clk
				askTo[bkt] = replyJoin.ReqIP
			}
		}
	}
	//"Invert" askTo, i.e., get the mapping of replica -> bkts.
	replicaToBkt := make(map[string][]string)
	for bkt, replica := range askTo {
		replicaToBkt[replica] = append(replicaToBkt[replica], bkt)
	}
	repl.waitFor = len(replicaToBkt)
	ownIP := localRabbitMQIP
	//Finally, send requestBkt msgs
	for askReplicaIP, buckets := range replicaToBkt {
		repl.remoteConn.SendRequestBucket(RequestBucket{Buckets: buckets, SenderID: repl.replicaID, ReqIP: ownIP}, askReplicaIP)
	}
}
